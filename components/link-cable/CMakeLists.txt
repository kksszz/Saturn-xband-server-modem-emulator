cmake_minimum_required(VERSION 3.20)
project(SaturnLinkCable LANGUAGES CXX)
add_library(saturn_link_cable STATIC link_cable_service.cpp)
target_compile_features(saturn_link_cable PUBLIC cxx_std_20)
target_include_directories(saturn_link_cable PUBLIC ${CMAKE_CURRENT_LIST_DIR})
if(WIN32)
  target_link_libraries(saturn_link_cable PUBLIC ws2_32)
endif()
include(CTest)
set(SATURN_YMIR_SOURCE "" CACHE PATH "Optional existing integrated YMIR source for wire interoperability tests")
if(BUILD_TESTING)
  add_executable(link-sci-test sci_state_test.cpp)
  set_target_properties(link-sci-test PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
  add_test(NAME link.sci_register_timing_dma_irq COMMAND link-sci-test)
  add_executable(link-transport-test transport_test.cpp)
  target_link_libraries(link-transport-test PRIVATE saturn_link_cable)
  add_test(NAME link.dual_sci_transport COMMAND link-transport-test)
  set_tests_properties(link.dual_sci_transport PROPERTIES TIMEOUT 25)
  if(SATURN_YMIR_SOURCE)
    set(ymir_service "${SATURN_YMIR_SOURCE}/apps/ymir-sdl3/src/app/services")
    if(NOT EXISTS "${ymir_service}/link_cable_service.cpp")
      message(FATAL_ERROR "SATURN_YMIR_SOURCE must be the integrated YMIR source")
    endif()
    add_executable(link-ymir-wire-test ymir_wire_test.cpp "${ymir_service}/link_cable_service.cpp")
    target_include_directories(link-ymir-wire-test PRIVATE "${ymir_service}" "${SATURN_YMIR_SOURCE}/libs/ymir-core/include")
    target_compile_definitions(link-ymir-wire-test PRIVATE SATURN_ORIGINAL_YMIR_LINK_HEADER="${ymir_service}/link_cable_service.hpp")
    target_link_libraries(link-ymir-wire-test PRIVATE saturn_link_cable)
    add_test(NAME link.ymir_wire_compatibility COMMAND link-ymir-wire-test)
    set_tests_properties(link.ymir_wire_compatibility PROPERTIES TIMEOUT 20)
  endif()
  if(WIN32)
    add_executable(link-mednafen-runtime-test
      ../xband/adapters/mednafen/cable_runtime.cpp
      mednafen_runtime_test.cpp)
    target_compile_definitions(link-mednafen-runtime-test PRIVATE XBAND_MEDNAFEN_LINK_CABLE=1)
    target_include_directories(link-mednafen-runtime-test PRIVATE ../xband/adapters/mednafen)
    target_link_libraries(link-mednafen-runtime-test PRIVATE saturn_link_cable)
    add_test(NAME link.mednafen_c_abi_lifecycle COMMAND link-mednafen-runtime-test)
    set_tests_properties(link.mednafen_c_abi_lifecycle PROPERTIES TIMEOUT 20)
  endif()
endif()
