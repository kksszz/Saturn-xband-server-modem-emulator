cmake_minimum_required(VERSION 3.20)
project(XbandMednafenAdapter LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This adapter currently requires Windows")
endif()
set(XBAND_JSON_INCLUDE_DIR "" CACHE PATH "Directory containing nlohmann/json.hpp")
if(NOT EXISTS "${XBAND_JSON_INCLUDE_DIR}/nlohmann/json.hpp")
  message(FATAL_ERROR "Set XBAND_JSON_INCLUDE_DIR to nlohmann/json 3.12.0 include directory")
endif()
get_filename_component(XBAND_COMPONENT_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
get_filename_component(XBAND_REPOSITORY_DIR "${XBAND_COMPONENT_DIR}/../.." ABSOLUTE)
add_library(xband_mednafen_runtime STATIC command_runtime.cpp card_runtime.cpp)
target_include_directories(xband_mednafen_runtime PUBLIC "${CMAKE_CURRENT_LIST_DIR}" PRIVATE "${XBAND_COMPONENT_DIR}/include" "${XBAND_JSON_INCLUDE_DIR}")
target_compile_features(xband_mednafen_runtime PRIVATE cxx_std_20)
target_link_libraries(xband_mednafen_runtime PUBLIC ws2_32 bcrypt iphlpapi)
if(MSVC)
  target_compile_options(xband_mednafen_runtime PRIVATE /utf-8)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(mednafen-bus-test bus_bridge_test.cpp)
  target_include_directories(mednafen-bus-test PRIVATE "${XBAND_COMPONENT_DIR}/include")
  target_compile_features(mednafen-bus-test PRIVATE cxx_std_20)
  add_test(NAME mednafen.requested_bus_contract COMMAND mednafen-bus-test)
  add_executable(mednafen-card-test card_runtime_test.cpp)
  target_include_directories(mednafen-card-test PRIVATE "${XBAND_COMPONENT_DIR}/include" "${XBAND_JSON_INCLUDE_DIR}")
  target_link_libraries(mednafen-card-test PRIVATE xband_mednafen_runtime)
  target_compile_features(mednafen-card-test PRIVATE cxx_std_20)
  add_test(NAME mednafen.card_runtime_contract COMMAND mednafen-card-test)
  # Full repository and v0.5.0 ZIP; smaller/older adapter-only exports omit this.
  if(EXISTS "${XBAND_REPOSITORY_DIR}/outputs/live-capture-tests/pb3_pair_control.hpp")
    add_executable(mednafen-command-test command_runtime_test.cpp)
    target_include_directories(mednafen-command-test PRIVATE "${XBAND_COMPONENT_DIR}/include" "${XBAND_JSON_INCLUDE_DIR}" "${XBAND_REPOSITORY_DIR}/outputs/live-capture-tests")
    target_link_libraries(mednafen-command-test PRIVATE xband_mednafen_runtime winmm)
    target_compile_features(mednafen-command-test PRIVATE cxx_std_20)
    if(MSVC)
      target_compile_options(mednafen-command-test PRIVATE /utf-8)
    endif()
    add_test(NAME mednafen.command_runtime_contract COMMAND mednafen-command-test)
  endif()
endif()
